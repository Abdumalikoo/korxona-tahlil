/// Komponent qaysi manbadan to'lanadi
enum SalaryComponentGroup {
  BUDGET   // Byudjet hisobidan
  HOJALIK  // Xo'jalik hisobidan
}

/// Tahlil uchun yiriklashtirilgan guruh
enum SalaryComponentCategory {
  ASOSIY         // Oklad, 40% ustama, staj
  TATIL          // Ta'til, kompensatsiya, kasallik, dekret
  PREMIYA        // KPI, bayram, bildirgi
  KOMPENSATSIYA  // Transport, ovqat, moddiy yordam
  USTAMA         // Sertifikat, unvon va boshqalar
  JARIMA         // Minus bilan hisoblanadi
  BOSHQA
}

/// Tarkib soliq hisobotiga nisbatan holati
enum SalaryCompositionStatus {
  MATCHED     // To'liq mos
  MISMATCH    // Farq bor
  NO_PAYROLL  // Shu oy uchun soliq hisoboti yuklanmagan
}

/// Ish haqi komponentlari ma'lumotnomasi
model SalaryComponent {
  code      String                  @id
  name      String
  group     SalaryComponentGroup
  category  SalaryComponentCategory
  /// 1 = qo'shiladi, -1 = ayiriladi (jarima)
  sign      Int                     @default(1)
  sortOrder Int                     @default(0)
  isActive  Boolean                 @default(true)
  createdAt DateTime                @default(now())
  updatedAt DateTime                @updatedAt

  items SalaryCompositionItem[]
}

/// Bir oylik tarkib yuklamasi
model SalaryCompositionBatch {
  id            String                  @id @default(cuid())
  /// "YYYY-MM"
  period        String
  fileName      String?
  employeeCount Int                     @default(0)
  /// Tarkib bo'yicha jami (jarima ayirilgan)
  totalTiyin    BigInt                  @default(0)
  /// Soliq hisobotidagi jami (yuklash paytida)
  payrollTiyin  BigInt                  @default(0)
  status        SalaryCompositionStatus @default(NO_PAYROLL)
  mismatchCount Int                     @default(0)
  uploadedById  String?
  createdAt     DateTime                @default(now())
  cancelledAt   DateTime?

  rows  SalaryCompositionRow[]
  items SalaryCompositionItem[]

  @@index([period])
}

/// Xodimning bir oylik tarkib qatori
model SalaryCompositionRow {
  id            String   @id @default(cuid())
  batchId       String
  period        String
  employeePinfl String
  fullName      String?
  workedDays    Float?
  rate          Float?
  totalTiyin    BigInt   @default(0)

  batch SalaryCompositionBatch  @relation(fields: [batchId], references: [id], onDelete: Cascade)
  items SalaryCompositionItem[]

  @@unique([batchId, employeePinfl])
  @@index([period, employeePinfl])
}

/// Xodim x komponent summasi (doim musbat, belgi komponentdan olinadi)
model SalaryCompositionItem {
  id            String @id @default(cuid())
  batchId       String
  rowId         String
  componentCode String
  amountTiyin   BigInt

  batch     SalaryCompositionBatch @relation(fields: [batchId], references: [id], onDelete: Cascade)
  row       SalaryCompositionRow   @relation(fields: [rowId], references: [id], onDelete: Cascade)
  component SalaryComponent        @relation(fields: [componentCode], references: [code])

  @@unique([rowId, componentCode])
  @@index([batchId, componentCode])
}

  private periodBounds(period: string): { from: string; to: string } {
    const [year, month] = period.split('-').map(Number);
    const lastDay = new Date(Date.UTC(year ?? 2026, month ?? 1, 0)).getUTCDate();

    return {
      from: `${period}-01`,
      to: `${period}-${String(lastDay).padStart(2, '0')}`,
    };
  }

  private periodEndDate(period: string): Date {
    const [year, month] = period.split('-').map(Number);
    const nextMonth = month === 12 ? 1 : (month ?? 1) + 1;
    const nextYear = month === 12 ? (year ?? 2026) + 1 : (year ?? 2026);

    return new Date(
      Date.UTC(nextYear, nextMonth - 1, 1, -TASHKENT_OFFSET_HOURS) - 86_400_000,

    if (value && typeof value === 'object') {

      if ('text' in record) return record.text;
      if ('richText' in record && Array.isArray(record.richText)) {
        return (record.richText as { text: string }[]).map((part) => part.text).join('');
      }
    }
    return value;
  }

    const raw = this.cellValue(value);

  private normalize(value: string): string {
    return value
      .toLowerCase()
      .replace(/[\u0027\u0060\u00B4\u02BB\u02BC\u2018\u2019]/g, '')
      .replace(/\s+/g, ' ')
      .trim();
  }

  private parseNumber(value: unknown): number | null {
    if (value === null || value === undefined || String(value).trim() === '') return null;
    if (typeof value === 'number') return Number.isFinite(value) && value >= 0 ? value : null;

    const text = String(value).replace(/[\s\u00A0\u202F]/g, '').replace(',', '.');
    const parsed = Number(text);

    return Number.isFinite(parsed) && parsed >= 0 ? parsed : null;

    const byShort = new Map(regions.map((r) => [this.shortKey(r.name), r]));

    const ignored = ['№', 'xizmat turi', 'korsatkich', 'shundan,', 'shundan'];

    let best = { row: 0, columns: new Map<number, RegionRef>(), unknown: [] as string[] };

    const limit = Math.min(sheet.rowCount, 15);

    for (let rowIndex = 1; rowIndex <= limit; rowIndex++) {
      const columns = new Map<number, RegionRef>();
      const unknown: string[] = [];
      const used = new Set<number>();

      sheet.getRow(rowIndex).eachCell((cell, col) => {
        const raw = this.text(cell.value);
        if (!raw) return;

        const region = byFull.get(this.normalize(raw)) ?? byShort.get(this.shortKey(raw));

        if (region && !used.has(region.code)) {
          columns.set(col, region);
          used.add(region.code);

  private parseMoney(value: unknown): bigint | null {

      // Turi — faqat 1 yoki 2
      row.getCell(6).dataValidation = {
        type: 'whole',
        operator: 'between',
        allowBlank: true,
        formulae: [1, 2],
        showErrorMessage: true,
        errorTitle: 'Xodim turi',
        error: '1 - asosiy shtat, 2 - shartnoma',
      };
      row.getCell(6).alignment = { horizontal: 'center' };

      row.getCell(9).value = {
        formula: `IF(OR(G${r}="",H${r}="",G${r}=0),"",ROUND(H${r}/G${r}*100,0))`,
      };
      row.getCell(10).value = {
        formula:
          `IF(OR(G${r}="",H${r}="",G${r}=0),"",` +
          `IF(H${r}/G${r}*100<${LOW_PERCENT},0,` +
          `IF(H${r}/G${r}*100<=${HIGH_PERCENT},H${r}*0.25,H${r}*0.45)))`,
      };

      row.getCell(7).numFmt = '#,##0';
      row.getCell(8).numFmt = '#,##0';
      row.getCell(10).numFmt = '#,##0';

      for (let col = 1; col <= 10; col++) {
        row.getCell(col).border = border;
      }
    }

    // ─── Kodlar varag'i ───
    const codes = workbook.addWorksheet('Kodlar', {
      views: [{ state: 'frozen', ySplit: 1 }],
    });

    codes.columns = [
      { header: 'Hudud kodi', key: 'regionCode', width: 12 },
      { header: 'Hudud', key: 'regionName', width: 30 },
      { header: 'Tuman kodi', key: 'districtCode', width: 12 },
      { header: 'Tuman', key: 'districtName', width: 34 },
    ];

    const codesHeader = codes.getRow(1);
    codesHeader.font = { bold: true, color: { argb: 'FFFFFFFF' } };
    codesHeader.fill = { type: 'pattern', pattern: 'solid', fgColor: { argb: 'FF475569' } };

    for (const region of regions) {
      const list = districts
        .filter((district) => district.regionCode === region.code)
        .sort((a, b) => a.code - b.code);

      if (list.length === 0) {
        codes.addRow({ regionCode: region.code, regionName: region.name });
        continue;
      }

      for (const district of list) {
        codes.addRow({
          regionCode: region.code,
          regionName: region.name,
          districtCode: district.code,
          districtName: district.name,
        });
      }
    }

    codes.autoFilter = { from: 'A1', to: `D${codes.rowCount}` };

    const data = await workbook.xlsx.writeBuffer();
    return { buffer: Buffer.from(data), filename: `Xodim-natijalari-${period}.xlsx` };
  }

  // ═══════════ Fayl o'qish ═══════════

  /** Sarlavha qatorini topib, ustunlarni matni bo'yicha aniqlaydi */
  private detectColumns(sheet: ExcelJS.Worksheet): { headerRow: number; columns: ColumnMap } {
    const limit = Math.min(sheet.rowCount, 10);

    for (let rowIndex = 1; rowIndex <= limit; rowIndex++) {
      const row = sheet.getRow(rowIndex);
      const columns: ColumnMap = {};

      row.eachCell((cell, col) => {
        const key = this.normalize(this.text(cell.value));
        if (!key) return;

        // Tartib muhim: "ustama" matnida "tushum" so'zi ham bor
        if (key === 'pinfl') columns.pinfl ??= col;
        else if (key.startsWith('ustama')) columns.bonus ??= col;
        else if (key.startsWith('hudud')) columns.region ??= col;
        else if (key.startsWith('tuman')) columns.district ??= col;
        else if (key.includes('f.i.sh') || key.includes('fish') || key.startsWith('xodim')) {
          columns.fullName ??= col;
        } else if (key.startsWith('turi') || key.includes('shtat') || key.includes('shartnoma')) {
          columns.type ??= col;
        } else if (key.includes('bajarilishi') || key.includes('tushum')) columns.fact ??= col;
        else if (key.includes('reja')) columns.plan ??= col;
      });

      if (!columns.pinfl) continue;

      // Turi sarlavhasiz bo'lsa — F.I.Sh dan keyingi bo'sh sarlavhali ustun
      if (!columns.type && columns.fullName) {
        const next = columns.fullName + 1;
        if (!this.text(row.getCell(next).value)) {
          columns.type = next;
        }
      }

      return { headerRow: rowIndex, columns };
    }

    throw new BadRequestException('"PINFL" sarlavhasi topilmadi. Shablon tuzilishini saqlang');
  }

  async analyze(period: string, file: Express.Multer.File) {
    this.assertPeriod(period);

    if (!file) {
      throw new BadRequestException('Fayl yuklanmadi');
    }

    const workbook = new ExcelJS.Workbook();
    await workbook.xlsx.load(file.buffer as unknown as ArrayBuffer);

    const sheet = workbook.worksheets[0];
    if (!sheet) {
      throw new BadRequestException('Faylda varaq topilmadi');
    }

    const { headerRow, columns } = this.detectColumns(sheet);

    const required: [ColumnKey, string][] = [
      ['region', 'Hudud'],
      ['plan', 'Oylik reja'],
      ['fact', 'Reja bajarilishi (tushum)'],
    ];

    for (const [key, label] of required) {
      if (!columns[key]) {
        throw new BadRequestException(`"${label}" ustuni topilmadi`);
      }
    }

    const { regions, districts } = await this.loadReferences();

    const regionByCode = new Map<number, RegionRef>(regions.map((r) => [r.code, r]));
    const regionByName = new Map<string, RegionRef>();
    for (const region of regions) {
      regionByName.set(this.normalize(region.name), region);
      regionByName.set(this.shortKey(region.name), region);
    }

    const districtById = new Map<string, DistrictRef>(districts.map((d) => [d.id, d]));

    const resolveRegion = (raw: string): RegionRef | null => {
      if (/^\d+$/.test(raw)) return regionByCode.get(Number(raw)) ?? null;
      return regionByName.get(this.normalize(raw)) ?? regionByName.get(this.shortKey(raw)) ?? null;
    };

    const resolveDistrict = (regionCode: number, raw: string): DistrictRef | null => {
      if (/^\d+$/.test(raw)) return districtById.get(`${regionCode}-${Number(raw)}`) ?? null;

      const full = this.normalize(raw);
      const short = this.shortKey(raw);

      return (
        districts.find(
          (d) =>
            d.regionCode === regionCode &&
            (this.normalize(d.name) === full || this.shortKey(d.name) === short),
        ) ?? null
      );
    };

    const invalidRows: { rowIndex: number; reason: string }[] = [];
    const unknownDistricts: { rowIndex: number; region: string; value: string }[] = [];
    const parsed: ParsedLine[] = [];
    const seen = new Set<string>();

    for (let rowIndex = headerRow + 1; rowIndex <= sheet.rowCount; rowIndex++) {
      const row = sheet.getRow(rowIndex);
      const cell = (key: ColumnKey) => {
        const col = columns[key];
        return col ? row.getCell(col).value : null;
      };

      const rawPinfl = this.cellValue(cell('pinfl'));

      // PINFL bo'sh — "Jami" yoki bo'sh qator
      if (this.isBlank(rawPinfl)) continue;

      const pinfl = this.parsePinfl(rawPinfl);
      if (!pinfl) {
        invalidRows.push({ rowIndex, reason: 'PINFL notogri' });
        continue;
      }

      if (seen.has(pinfl)) {
        invalidRows.push({ rowIndex, reason: `PINFL takrorlangan: ${pinfl}` });
        continue;
      }

      const regionText = this.text(cell('region'));
      const region = resolveRegion(regionText);

      if (!region) {
        invalidRows.push({ rowIndex, reason: `Hudud topilmadi: "${regionText}"` });
        continue;
      }

      let district: DistrictRef | null = null;
      const districtText = this.text(cell('district'));

      if (districtText) {
        district = resolveDistrict(region.code, districtText);
        if (!district) {
          unknownDistricts.push({ rowIndex, region: region.name, value: districtText });
        }
      }

      const fileType = this.parseType(this.cellValue(cell('type')));
      if (fileType === 'invalid') {
        invalidRows.push({ rowIndex, reason: 'Turi notogri: 1 yoki 2 bolishi kerak' });
        continue;
      }

      const planTiyin = this.parseMoney(this.cellValue(cell('plan')));
      if (planTiyin === null) {
        invalidRows.push({ rowIndex, reason: 'Oylik reja notogri' });
        continue;
      }

      const factTiyin = this.parseMoney(this.cellValue(cell('fact')));
      if (factTiyin === null) {
        invalidRows.push({ rowIndex, reason: 'Tushum notogri' });
        continue;
      }

      let fileBonusTiyin: bigint | null = null;
      if (columns.bonus) {
        const bonus = this.parseMoney(this.cellValue(cell('bonus')));
        if (bonus === null) {
          invalidRows.push({ rowIndex, reason: 'Ustama notogri' });
          continue;
        }
        fileBonusTiyin = bonus;
      }

      seen.add(pinfl);
      parsed.push({
        rowIndex,
        pinfl,
        fileName: this.text(cell('fullName')),
        region,
        district,
        fileType,
        planTiyin,
        factTiyin,
        fileBonusTiyin,
      });
    }

    if (parsed.length === 0) {
      throw new BadRequestException('Faylda yaroqli qator topilmadi');
    }

    // ─── Reestr bilan solishtirish ───
    const employees = await this.prisma.employee.findMany({
      where: { pinfl: { in: parsed.map((row) => row.pinfl) } },
      select: {
        pinfl: true,
        fullName: true,
        regionCode: true,
        districtId: true,
        employmentType: true,
        region: { select: { name: true } },
        district: { select: { name: true } },
      },
    });

    type RegistryEmployee = (typeof employees)[number];
    const employeeMap = new Map<string, RegistryEmployee>(employees.map((e) => [e.pinfl, e]));

    const missing: { rowIndex: number; pinfl: string; fullName: string }[] = [];
    const nameMismatches: { rowIndex: number; pinfl: string; fileName: string; registryName: string }[] = [];
    const locationChanges: { pinfl: string; fullName: string; from: string; to: string }[] = [];
    const typeChanges: { pinfl: string; fullName: string; from: string; to: string }[] = [];
    const bonusMismatches: {
      rowIndex: number;
      pinfl: string;
      fullName: string;
      percent: number | null;
      fileTiyin: bigint;
      expectedTiyin: bigint;
    }[] = [];

    const rows: AnalyzedRow[] = [];

    const describe = (region: string, district: string | null | undefined) =>
      district ? `${region} / ${district}` : region;

    for (const line of parsed) {
      const employee = employeeMap.get(line.pinfl);

      if (!employee) {
        missing.push({ rowIndex: line.rowIndex, pinfl: line.pinfl, fullName: line.fileName });
        continue;
      }

      if (line.fileName && this.normalize(line.fileName) !== this.normalize(employee.fullName)) {
        nameMismatches.push({
          rowIndex: line.rowIndex,
          pinfl: line.pinfl,
          fileName: line.fileName,
          registryName: employee.fullName,
        });
      }

      const regionChanged = employee.regionCode !== line.region.code;
      const districtChanged = line.district !== null && employee.districtId !== line.district.id;

      if (regionChanged || districtChanged) {
        locationChanges.push({
          pinfl: line.pinfl,
          fullName: employee.fullName,
          from: describe(employee.region.name, employee.district?.name),
          to: describe(line.region.name, line.district?.name),
        });
      }

      const registryType = employee.employmentType as EmploymentKind;
      const employmentType = line.fileType ?? registryType;

      if (line.fileType && line.fileType !== registryType) {
        typeChanges.push({
          pinfl: line.pinfl,
          fullName: employee.fullName,
          from: TYPE_LABELS[registryType] ?? registryType,
          to: TYPE_LABELS[line.fileType],
        });
      }

      const expected = this.expectedBonus(line.planTiyin, line.factTiyin);
      const bonus = line.fileBonusTiyin ?? expected;
      const percent = this.percentOf(line.planTiyin, line.factTiyin);

      if (line.fileBonusTiyin !== null) {
        const diff = line.fileBonusTiyin - expected;
        if ((diff < 0n ? -diff : diff) > BONUS_TOLERANCE_TIYIN) {
          bonusMismatches.push({
            rowIndex: line.rowIndex,
            pinfl: line.pinfl,
            fullName: employee.fullName,
            percent,
            fileTiyin: line.fileBonusTiyin,
            expectedTiyin: expected,
          });
        }
      }

      rows.push({
        rowIndex: line.rowIndex,
        pinfl: line.pinfl,
        fullName: employee.fullName,
        regionCode: line.region.code,
        regionName: line.region.name,
        districtId: line.district?.id ?? null,
        districtName: line.district?.name ?? null,
        employmentType,
        planTiyin: line.planTiyin,
        factTiyin: line.factTiyin,
        percent,
        bonusTiyin: bonus,
        expectedBonusTiyin: expected,
      });
    }

    const regionTotals = this.buildRegionTotals(rows);

    const totals = rows.reduce(
      (acc, row) => ({
        planTiyin: acc.planTiyin + row.planTiyin,
        factTiyin: acc.factTiyin + row.factTiyin,
        bonusTiyin: acc.bonusTiyin + row.bonusTiyin,
      }),
      { planTiyin: 0n, factTiyin: 0n, bonusTiyin: 0n },
    );

    const previousCount = await this.prisma.resultBatch.count({
      where: { period, cancelledAt: null },
    });

    return {
      period,
      fileName: file.originalname,
      totalRows: parsed.length + invalidRows.length,
      matchedRows: rows.length,
      planTiyin: totals.planTiyin,
      factTiyin: totals.factTiyin,
      bonusTiyin: totals.bonusTiyin,
      percent: this.percentOf(totals.planTiyin, totals.factTiyin),
      invalidRows,
      missing,
      unknownDistricts,
      nameMismatches,
      locationChanges,
      typeChanges,
      bonusMismatches,
      hasPrevious: previousCount > 0,
      regions: regionTotals,
      rows,
    };
  }

  /** Hududlar bo'yicha yig'indi — hisobot tartibida */
  private buildRegionTotals(
    rows: { regionCode: number; regionName: string; planTiyin: bigint; factTiyin: bigint; bonusTiyin: bigint }[],
  ): RegionTotal[] {
    const map = new Map<number, RegionTotal>();

    for (const row of rows) {
      const item = map.get(row.regionCode) ?? {
        regionCode: row.regionCode,
        name: row.regionName,
        count: 0,
        planTiyin: 0n,
        factTiyin: 0n,
        bonusTiyin: 0n,
        percent: null,
      };

      item.count += 1;
      item.planTiyin += row.planTiyin;
      item.factTiyin += row.factTiyin;
      item.bonusTiyin += row.bonusTiyin;
      map.set(row.regionCode, item);
    }

    return [...map.values()]
      .map((item) => ({ ...item, percent: this.percentOf(item.planTiyin, item.factTiyin) }))
      .sort((a, b) => this.regionPosition(a.regionCode) - this.regionPosition(b.regionCode));
  }

  // ═══════════ Saqlash ═══════════

  /**
   * Natijalarni saqlaydi va shu oy uchun hudud tarixini yozadi.
   * Shu oyning oldingi yuklashi bekor qilinadi — natijalar oylik surat.
   */
  async commit(params: { period: string; rows: ResultRow[]; fileName: string; userId: string }) {
    const { period, rows, fileName, userId } = params;
    this.assertPeriod(period);

    if (rows.length === 0) {
      throw new BadRequestException('Saqlash uchun malumot yoq');
    }

    const pinfls = new Set<string>();

    for (const row of rows) {
      if (!/^\d{14}$/.test(row.pinfl)) {
        throw new BadRequestException(`Notogri PINFL: ${row.pinfl}`);
      }
      if (pinfls.has(row.pinfl)) {
        throw new BadRequestException(`PINFL takrorlangan: ${row.pinfl}`);
      }
      if (row.employmentType !== 'SHTAT' && row.employmentType !== 'SHARTNOMA') {
        throw new BadRequestException(`Xodim turi notogri: ${row.pinfl}`);
      }
      if (row.planTiyin < 0n || row.factTiyin < 0n || row.bonusTiyin < 0n) {
        throw new BadRequestException(`Manfiy summa: ${row.pinfl}`);
      }
      pinfls.add(row.pinfl);
    }

    // Reestr va ma'lumotnomalarni qayta tekshiramiz — frontendga ishonmaymiz
    const [employees, { regions, districts }] = await Promise.all([
      this.prisma.employee.findMany({
        where: { pinfl: { in: [...pinfls] } },
        select: { pinfl: true, regionCode: true, districtId: true, employmentType: true },
      }),
      this.loadReferences(),
    ]);

    if (employees.length !== pinfls.size) {
      throw new BadRequestException(
        `${pinfls.size - employees.length} ta xodim reestrda topilmadi. Faylni qaytadan yuklang`,
      );
    }

    const regionCodes = new Set(regions.map((r) => r.code));
    const districtRegion = new Map<string, number>(districts.map((d) => [d.id, d.regionCode]));

    for (const row of rows) {
      if (!regionCodes.has(row.regionCode)) {
        throw new BadRequestException(`Hudud topilmadi: ${row.regionCode}`);
      }
      if (row.districtId && districtRegion.get(row.districtId) !== row.regionCode) {
        throw new BadRequestException(`Tuman hududga mos emas: ${row.districtId}`);
      }
    }

    const registry = new Map(employees.map((e) => [e.pinfl, e]));

    const movedCount = rows.filter((row) => {
      const employee = registry.get(row.pinfl);
      if (!employee) return false;

      return (
        employee.regionCode !== row.regionCode ||
        (row.districtId !== null && employee.districtId !== row.districtId) ||
        employee.employmentType !== row.employmentType
      );
    }).length;

    const totals = rows.reduce(
      (acc, row) => ({
        planTiyin: acc.planTiyin + row.planTiyin,
        factTiyin: acc.factTiyin + row.factTiyin,
        bonusTiyin: acc.bonusTiyin + row.bonusTiyin,
      }),
      { planTiyin: 0n, factTiyin: 0n, bonusTiyin: 0n },
    );

    const result = await this.prisma.$transaction(
      async (tx) => {
        // Shu oyning oldingi yuklashi bekor qilinadi
        const previous = await tx.resultBatch.findMany({
          where: { period, cancelledAt: null },
          select: { id: true },
        });
        const previousIds = previous.map((item) => item.id);

        if (previousIds.length > 0) {
          await tx.employeeResult.deleteMany({ where: { batchId: { in: previousIds } } });
          await tx.resultBatch.updateMany({
            where: { id: { in: previousIds } },
            data: { cancelledAt: new Date() },
          });
        }

        // Shu oy hudud tarixi qaytadan yoziladi
        await tx.employeeLocation.deleteMany({ where: { period, source: 'RESULTS' } });

        const batch = await tx.resultBatch.create({
          data: {
            period,
            fileName,
            totalRows: rows.length,
            planTiyin: totals.planTiyin,
            factTiyin: totals.factTiyin,
            bonusTiyin: totals.bonusTiyin,
            movedCount,
            uploadedById: userId,
          },
        });

        await tx.employeeResult.createMany({
          data: rows.map((row) => ({
            batchId: batch.id,
            period,
            employeePinfl: row.pinfl,
            regionCode: row.regionCode,
            districtId: row.districtId,
            employmentType: row.employmentType,
            planTiyin: row.planTiyin,
            factTiyin: row.factTiyin,
            bonusTiyin: row.bonusTiyin,
          })),
        });

        await tx.employeeLocation.createMany({
          data: rows.map((row) => ({
            employeePinfl: row.pinfl,
            period,
            regionCode: row.regionCode,
            districtId: row.districtId,
            employmentType: row.employmentType,
            source: 'RESULTS',
          })),
        });

        return { batchId: batch.id, replaced: previousIds.length };
      },
      { timeout: 60_000 },
    );

    // Ish haqi yangi hudud va turga qarab qayta guruhlanadi

    const payroll = await this.payroll.regroup(period);


    return {
      success: true as const,
      batchId: result.batchId,
      payroll,
      saved: rows.length,
      movedCount,
      replacedPrevious: result.replaced > 0,
      ...totals,
    };
  }

  // ═══════════ Bekor qilish ═══════════

  async cancel(batchId: string) {
    const batch = await this.prisma.resultBatch.findUnique({
      where: { id: batchId },
      include: { results: { select: { employeePinfl: true } } },
    });

    if (!batch) {
      throw new NotFoundException('Yuklash topilmadi');
    }
    if (batch.cancelledAt) {
      throw new BadRequestException('Bu yuklash allaqachon bekor qilingan');
    }

    const pinfls = batch.results.map((item) => item.employeePinfl);

    await this.prisma.$transaction([
      this.prisma.employeeLocation.deleteMany({
        where: { period: batch.period, source: 'RESULTS', employeePinfl: { in: pinfls } },
      }),
      this.prisma.employeeResult.deleteMany({ where: { batchId } }),
      this.prisma.resultBatch.update({
        where: { id: batchId },
        data: { cancelledAt: new Date() },
      }),
    ]);

    const payroll = await this.payroll.regroup(batch.period);
    return { success: true as const, payroll };
  }

  // ═══════════ Ro'yxatlar ═══════════

  async findBatches(period?: string) {
    return this.prisma.resultBatch.findMany({
      where: period ? { period } : {},
      orderBy: { createdAt: 'desc' },
      take: 50,
      include: { uploadedBy: { select: { id: true, fullName: true } } },
    });
  }

  /** Oy natijalari — hudud, tuman, xodim bo'yicha */
  async findResults(period: string, regionCode?: number) {
    this.assertPeriod(period);

    const results = await this.prisma.employeeResult.findMany({
      where: {
        period,
        batch: { cancelledAt: null },
        ...(regionCode !== undefined ? { regionCode } : {}),
      },
      include: {
        employee: { select: { fullName: true } },
        region: { select: { name: true } },
        district: { select: { name: true } },
      },
    });

    const rows = results
      .map((item) => ({
        pinfl: item.employeePinfl,
        fullName: item.employee.fullName,
        regionCode: item.regionCode,
        regionName: item.region.name,
        districtId: item.districtId,
        districtName: item.district?.name ?? null,
        employmentType: item.employmentType as EmploymentKind,
        planTiyin: item.planTiyin,
        factTiyin: item.factTiyin,
        bonusTiyin: item.bonusTiyin,
        percent: this.percentOf(item.planTiyin, item.factTiyin),
      }))
      .sort(
        (a, b) =>
          this.regionPosition(a.regionCode) - this.regionPosition(b.regionCode) ||
          (a.districtName ?? '').localeCompare(b.districtName ?? '') ||
          a.fullName.localeCompare(b.fullName),
      );

    const totals = rows.reduce(
      (acc, row) => ({
        planTiyin: acc.planTiyin + row.planTiyin,
        factTiyin: acc.factTiyin + row.factTiyin,
        bonusTiyin: acc.bonusTiyin + row.bonusTiyin,
      }),
      { planTiyin: 0n, factTiyin: 0n, bonusTiyin: 0n },
    );

    return {
      rows,
      regions: this.buildRegionTotals(rows),
      totals: { ...totals, percent: this.percentOf(totals.planTiyin, totals.factTiyin) },
    };
  }

  // ═══════════ Excel eksport ═══════════

  /** Oy natijalari — xodimlar va hududlar varag'i, shablon ko'rinishida */
  async exportExcel(period: string, regionCode?: number): Promise<{ buffer: Buffer; filename: string }> {
    const { rows, regions, totals } = await this.findResults(period, regionCode);

    if (rows.length === 0) {

  private async buildCheck(
    period: string,
    rows: CheckInput[] | null,
    payroll: PayrollInfo,
  ): Promise<CheckSummary> {
    const compositionUploaded = rows !== null;
    const list = rows ?? [];

    let compositionTotal = 0n;
    let payrollTotal = 0n;
    for (const value of payroll.map.values()) payrollTotal += value;

    const mismatches: CheckPerson[] = [];
    const notInPayroll: CheckPerson[] = [];
    const inComposition = new Set<string>();

    for (const row of list) {
      compositionTotal += row.totalTiyin;
      inComposition.add(row.pinfl);
      if (!payroll.uploaded) continue;

      const payrollTiyin = payroll.map.get(row.pinfl);

      if (payrollTiyin === undefined) {
        if (row.totalTiyin !== 0n) {
          notInPayroll.push({
            pinfl: row.pinfl,
            fullName: row.fullName,
            compositionTiyin: row.totalTiyin.toString(),
            payrollTiyin: '0',
            diffTiyin: row.totalTiyin.toString(),
          });
        }
        continue;
      }

      const diff = row.totalTiyin - payrollTiyin;
      if (this.abs(diff) > DIFF_TOLERANCE_TIYIN) {
        mismatches.push({
          pinfl: row.pinfl,
          fullName: row.fullName,
          compositionTiyin: row.totalTiyin.toString(),
          payrollTiyin: payrollTiyin.toString(),
          diffTiyin: diff.toString(),
        });
      }
    }

    const missingInComposition: CheckPerson[] = [];
    if (payroll.uploaded && compositionUploaded) {
      const absent = [...payroll.map.entries()].filter(
        ([pinfl, amount]) => amount !== 0n && !inComposition.has(pinfl),
      );
      const names = await this.registry(absent.map(([pinfl]) => pinfl));

      for (const [pinfl, amount] of absent) {
        missingInComposition.push({
          pinfl,
          fullName: names.get(pinfl)?.fullName ?? '—',
          compositionTiyin: '0',
          payrollTiyin: amount.toString(),
          diffTiyin: (-amount).toString(),
        });
      }
    }

    const byDiff = (a: CheckPerson, b: CheckPerson) =>
      this.abs(BigInt(b.diffTiyin)) > this.abs(BigInt(a.diffTiyin)) ? 1 : -1;
    mismatches.sort(byDiff);
    missingInComposition.sort(byDiff);
    notInPayroll.sort(byDiff);

    const mismatchCount = mismatches.length + missingInComposition.length + notInPayroll.length;

    let state: CheckState;
    if (!compositionUploaded) state = 'NOT_UPLOADED';
    else if (!payroll.uploaded) state = 'NO_PAYROLL';
    else state = mismatchCount === 0 ? 'MATCHED' : 'MISMATCH';

    return {
      period,
      state,
      payrollUploaded: payroll.uploaded,
      compositionUploaded,
      compositionTiyin: compositionTotal.toString(),
      payrollTiyin: payrollTotal.toString(),
      diffTiyin: (compositionTotal - payrollTotal).toString(),
      employees: list.length,
      mismatchCount,
      mismatches,
      missingInComposition,
      notInPayroll,
    };
  }

  private totals(
    rows: { items: Map<string, bigint> }[],
    components: SalaryComponent[],
  ): { byComponent: ComponentTotal[]; byCategory: CategoryTotal[] } {
    const sums = new Map<string, { total: bigint; employees: number }>();

    for (const row of rows) {
      for (const [code, amount] of row.items) {
        if (amount === 0n) continue;
        const current = sums.get(code) ?? { total: 0n, employees: 0 };
        current.total += amount;
        current.employees += 1;
        sums.set(code, current);
      }
    }

    const byComponent: ComponentTotal[] = [];
    const categorySums = new Map<SalaryComponentCategory, bigint>();
    let grand = 0n;

    for (const component of components) {
      const sum = sums.get(component.code);
      if (!sum) continue;

      const signed = component.sign < 0 ? -sum.total : sum.total;
      grand += signed;
      categorySums.set(component.category, (categorySums.get(component.category) ?? 0n) + signed);

      byComponent.push({
        code: component.code,
        name: component.name,
        group: component.group,
        category: component.category,
        sign: component.sign,
        totalTiyin: signed.toString(),
        employees: sum.employees,
      });
    }

    const byCategory: CategoryTotal[] = CATEGORY_ORDER.filter((category) =>
      categorySums.has(category),
    ).map((category) => {
      const total = categorySums.get(category) ?? 0n;
      return {
        category,
        label: CATEGORY_LABELS[category],
        totalTiyin: total.toString(),
        sharePercent: this.percent(this.abs(total), grand),
      };
    });

    return { byComponent, byCategory };
  }

  async createComponent(input: ComponentInput): Promise<ComponentDto> {
    const code = String(input.code ?? '')
      .trim()
      .toUpperCase();

    if (!/^[A-Z][A-Z0-9_]{1,29}$/.test(code)) {
      throw new BadRequestException(
        'Kod lotin katta harflar, raqam va "_" dan iborat bo\'lsin (2–30 belgi), masalan: TUNGI_ISH',
      );
    }
    if (RESERVED_CODES.includes(code)) {
      throw new BadRequestException(`"${code}" kodi band — boshqa kod tanlang`);
    }

    const exists = await this.prisma.salaryComponent.findUnique({ where: { code } });
    if (exists) throw new BadRequestException(`"${code}" kodli komponent allaqachon bor`);

    const name = this.parseName(input.name);
    const group = this.parseGroup(input.group);
    const category = this.parseCategory(input.category);
    const sign = category === 'JARIMA' || input.sign === -1 ? -1 : 1;

    let sortOrder = Number(input.sortOrder);
    if (!Number.isInteger(sortOrder)) {
      const last = await this.prisma.salaryComponent.aggregate({
        where: { group },
        _max: { sortOrder: true },
      });
      sortOrder = (last._max.sortOrder ?? 0) + 10;
    }

    const created = await this.prisma.salaryComponent.create({
      data: { code, name, group, category, sign, sortOrder, isActive: true },

    if (input.sortOrder !== undefined) {
      const sortOrder = Number(input.sortOrder);
      if (!Number.isInteger(sortOrder)) throw new BadRequestException('Tartib raqami butun son bo\'lsin');
      data.sortOrder = sortOrder;
    }
    if (input.isActive !== undefined) data.isActive = Boolean(input.isActive);

    const updated = await this.prisma.salaryComponent.update({ where: { code }, data });
    return this.toComponentDto(updated, usage);
  }

  async removeComponent(code: string): Promise<{ success: true }> {
    const current = await this.prisma.salaryComponent.findUnique({ where: { code } });
    if (!current) throw new NotFoundException('Komponent topilmadi');

    const usage = await this.prisma.salaryCompositionItem.count({ where: { componentCode: code } });
    if (usage > 0) {
      throw new BadRequestException(
        `Bu komponent ${usage} ta yozuvda ishlatilgan — o'chirib bo'lmaydi. Uni "nofaol" qiling, shablonda chiqmaydi.`,
      );
    }

    await this.prisma.salaryComponent.delete({ where: { code } });
    return { success: true };
  }

  async buildTemplate(period: string): Promise<{ buffer: Buffer; filename: string }> {
    this.assertPeriod(period);

    const components = (await this.allComponents()).filter((c) => c.isActive);
    if (components.length === 0) {
      throw new BadRequestException('Faol komponent yo\'q — Sozlamalarda komponent qo\'shing');
    }

    const payroll = await this.payroll(period);

    // Xodimlar varag'i: faol xodimlar + shu oy ish haqi olganlar
    const employees = await this.prisma.employee.findMany({
      where: {
        OR: [{ isActive: true }, { pinfl: { in: [...payroll.map.keys()] } }],
      },
      select: {
        pinfl: true,
        fullName: true,
        regionCode: true,
        region: { select: { name: true } },
        department: { select: { name: true } },
      },
      orderBy: { fullName: 'asc' },
    });

    const workbook = buildTemplateWorkbook({ period, components, employees, payroll });

    const buffer = Buffer.from(await workbook.xlsx.writeBuffer());
    return { buffer, filename: `ish-haqi-tarkibi-${period}.xlsx` };
  }

  // ═══════════════════════════════════════════════════════════
  // Faylni o'qish
  // ═══════════════════════════════════════════════════════════

  async analyze(period: string, file: Express.Multer.File | undefined): Promise<AnalysisResult> {
    this.assertPeriod(period);
    if (!file) throw new BadRequestException('Fayl yuklanmadi');

    const workbook = new ExcelJS.Workbook();
    try {
      await workbook.xlsx.load(file.buffer as unknown as ExcelJS.Buffer);
    } catch {
      throw new BadRequestException('Faylni o\'qib bo\'lmadi — .xlsx formatida saqlang');
    }

    const sheet = workbook.getWorksheet('Tarkib') ?? workbook.worksheets[0];
    if (!sheet) throw new BadRequestException('Faylda varaq yo\'q');

    const components = await this.allComponents();
    const signs = new Map(components.map((c) => [c.code, c.sign]));
    const byCode = new Map(components.map((c) => [c.code, c]));

    const { headerRow, fixed, comps } = detectColumns(sheet, components);

    const invalidRows: AnalysisResult['invalidRows'] = [];
    const duplicates: AnalysisResult['duplicates'] = [];
    const totalMismatches: AnalysisResult['totalMismatches'] = [];
    const parsed: ParsedRow[] = [];
    const seen = new Set<string>();
    let totalRows = 0;

    const pinflCol = fixed.pinfl as number;

    for (let r = headerRow + 1; r <= sheet.rowCount; r++) {
      const row = sheet.getRow(r);
      const pinflRaw = cellText(row.getCell(pinflCol).value);

      // Kodlar qatori (nomlar bo'yicha topilgan bo'lsa) yoki JAMI qatori — o'tkazamiz
      const upper = pinflRaw.toUpperCase();
      if (upper === 'JAMI' || upper === SERVICE_CODES.pinfl) continue;

      // Summalarni o'qiymiz
      const items = new Map<string, bigint>();
      const errors: string[] = [];
      let hasAmount = false;

      for (const [code, col] of comps) {
        const component = byCode.get(code);
        if (!component) continue;

        const value = parseNumber(row.getCell(col).value);
        if (value === null || value === 0) continue;

        if (Number.isNaN(value)) {
          errors.push(`${component.name}: son emas`);
          continue;
        }

        let amount = value;
        if (amount < 0) {
          if (component.sign < 0) amount = -amount;
          else {
            errors.push(`${component.name}: manfiy summa`);
            continue;
          }
        }

        const tiyin = toTiyin(amount);
        if (tiyin !== 0n) {
          items.set(code, tiyin);
          hasAmount = true;
        }
      }

      if (!pinflRaw) {
        if (hasAmount) {
          totalRows += 1;
          invalidRows.push({ rowIndex: r, reason: 'PINFL yozilmagan, lekin summa bor' });
        }
        continue;
      }

      totalRows += 1;
      const pinfl = pinflRaw.replace(/\D/g, '');

      if (pinfl.length !== 14) {
        invalidRows.push({ rowIndex: r, reason: `PINFL noto'g'ri: "${pinflRaw}" (14 ta raqam bo'lishi kerak)` });
        continue;
      }
      if (errors.length > 0) {
        invalidRows.push({ rowIndex: r, reason: errors.join('; ') });
        continue;
      }
      if (seen.has(pinfl)) {
        duplicates.push({ rowIndex: r, pinfl });
        continue;
      }
      seen.add(pinfl);

      const days = fixed.days ? parseNumber(row.getCell(fixed.days).value) : null;
      const rate = fixed.rate ? parseNumber(row.getCell(fixed.rate).value) : null;

      let fileTotalTiyin: bigint | null = null;
      if (fixed.total) {
        const total = parseNumber(row.getCell(fixed.total).value);
        if (total !== null && !Number.isNaN(total)) fileTotalTiyin = toTiyin(total);
      }

      const computed = this.rowTotal(items, signs);
      if (fileTotalTiyin !== null && this.abs(fileTotalTiyin - computed) > DIFF_TOLERANCE_TIYIN) {
        totalMismatches.push({
          rowIndex: r,
          pinfl,
          fileTiyin: fileTotalTiyin.toString(),
          computedTiyin: computed.toString(),
        });
      }

      parsed.push({
        rowIndex: r,
        pinfl,
        fileName: fixed.fullName ? cellText(row.getCell(fixed.fullName).value) : '',
        workedDays: days === null || Number.isNaN(days) ? null : days,
        rate: rate === null || Number.isNaN(rate) ? null : rate,
        items,
        fileTotalTiyin,
      });
    }

    if (parsed.length === 0 && invalidRows.length === 0) {
      throw new BadRequestException('Faylda xodim qatorlari topilmadi');
    }

    const [registry, payroll, previous] = await Promise.all([
      this.registry(parsed.map((p) => p.pinfl)),
      this.payroll(period),
      this.activeBatch(period),
    ]);

    const unknownEmployees: AnalysisResult['unknownEmployees'] = [];

    const rows: CompositionRowDto[] = parsed.map((p) => {
      const reg = registry.get(p.pinfl);
      if (!reg) unknownEmployees.push({ rowIndex: p.rowIndex, pinfl: p.pinfl, fileName: p.fileName });

      const total = this.rowTotal(p.items, signs);
      const state = this.rowState(p.pinfl, total, payroll);

      return {
        rowIndex: p.rowIndex,
        pinfl: p.pinfl,
        fullName: reg?.fullName ?? (p.fileName && p.fileName !== "Reestrda yo'q" ? p.fileName : '—'),
        inRegistry: !!reg,
        unitName: reg?.unitName ?? null,
        workedDays: p.workedDays,
        rate: p.rate,
        items: this.mapToRecord(p.items),
        totalTiyin: total.toString(),
        payrollTiyin: state.payrollTiyin === null ? null : state.payrollTiyin.toString(),
        diffTiyin: state.diffTiyin === null ? null : state.diffTiyin.toString(),
        state: state.state,
      };
    });

    const check = await this.buildCheck(
      period,
      rows.map((r) => ({ pinfl: r.pinfl, fullName: r.fullName, totalTiyin: BigInt(r.totalTiyin) })),
      payroll,
    );

    const found = components.filter((c) => comps.has(c.code));
    const missingColumns = components
      .filter((c) => c.isActive && !comps.has(c.code))
      .map((c) => ({ code: c.code, name: c.name }));

    const { byComponent, byCategory } = this.totals(parsed, components);

    return {
      period,
      fileName: file.originalname,
      totalRows,
      validRows: rows.length,
      columns: found.map((c) => this.toComponentDto(c)),
      missingColumns,
      invalidRows,
      duplicates,
      unknownEmployees,
      totalMismatches,
      hasPrevious: !!previous,
      byComponent,
      byCategory,
      check,
      rows,
    };
  }

  // ═══════════════════════════════════════════════════════════
  // Saqlash / bekor qilish
  // ═══════════════════════════════════════════════════════════

  async commit(input: {
    period: string;
    fileName: string;
    userId: string;
    rows: CommitRow[];
  }): Promise<CommitResult> {
    const { period, rows } = input;
    this.assertPeriod(period);

    if (rows.length === 0) throw new BadRequestException('Saqlash uchun ma\'lumot yo\'q');
    if (rows.length > 20000) throw new BadRequestException('Qatorlar juda ko\'p');

    const components = await this.allComponents();
    const signs = new Map(components.map((c) => [c.code, c.sign]));

    // ─── Tekshiruv ───
    const seen = new Set<string>();
    for (const row of rows) {
      if (!/^\d{14}$/.test(row.pinfl)) {
        throw new BadRequestException(`PINFL noto'g'ri: ${row.pinfl}`);
      }
      if (seen.has(row.pinfl)) throw new BadRequestException(`PINFL takrorlangan: ${row.pinfl}`);
      seen.add(row.pinfl);

      for (const [code, amount] of row.items) {
        if (!signs.has(code)) throw new BadRequestException(`Noma'lum komponent: ${code}`);
        if (amount < 0n) throw new BadRequestException(`Manfiy summa: ${row.pinfl} / ${code}`);
      }
    }

    // ─── Hisob ───
    const registry = await this.registry(rows.map((r) => r.pinfl));
    const prepared = rows.map((row) => ({
      id: randomUUID(),
      row,
      fullName: registry.get(row.pinfl)?.fullName ?? row.fullName ?? null,
      total: this.rowTotal(row.items, signs),
    }));

    const grandTotal = prepared.reduce((sum, p) => sum + p.total, 0n);
    const itemCount = prepared.reduce(
      (sum, p) => sum + [...p.row.items.values()].filter((a) => a !== 0n).length,
      0,
    );

    const payroll = await this.payroll(period);
    const check = await this.buildCheck(
      period,
      prepared.map((p) => ({ pinfl: p.row.pinfl, fullName: p.fullName ?? '—', totalTiyin: p.total })),
      payroll,
    );
    const status: SalaryCompositionStatus =
      check.state === 'MATCHED' ? 'MATCHED' : check.state === 'MISMATCH' ? 'MISMATCH' : 'NO_PAYROLL';

    // ─── Tranzaksiya ───
    const result = await this.prisma.$transaction(
      async (tx) => {
        const previous = await tx.salaryCompositionBatch.findMany({
          where: { period, cancelledAt: null },
          select: { id: true },
        });
        const previousIds = previous.map((p) => p.id);

        if (previousIds.length > 0) {
          await tx.salaryCompositionItem.deleteMany({ where: { batchId: { in: previousIds } } });
          await tx.salaryCompositionRow.deleteMany({ where: { batchId: { in: previousIds } } });
          await tx.salaryCompositionBatch.updateMany({
            where: { id: { in: previousIds } },
            data: { cancelledAt: new Date() },
          });
        }

        const batch = await tx.salaryCompositionBatch.create({
          data: {
            period,
            fileName: input.fileName.slice(0, 255),
            employeeCount: prepared.length,
            totalTiyin: grandTotal,
            payrollTiyin: BigInt(check.payrollTiyin),
            status,
            mismatchCount: check.mismatchCount,
            uploadedById: input.userId,
          },
        });

        await tx.salaryCompositionRow.createMany({
          data: prepared.map((p) => ({
            id: p.id,
            batchId: batch.id,
            period,
            employeePinfl: p.row.pinfl,
            fullName: p.fullName,
            workedDays: p.row.workedDays,
            rate: p.row.rate,
            totalTiyin: p.total,
          })),
        });

        const items = prepared.flatMap((p) =>
          [...p.row.items.entries()]
            .filter(([, amount]) => amount !== 0n)
            .map(([componentCode, amountTiyin]) => ({
              batchId: batch.id,
              rowId: p.id,
              componentCode,
              amountTiyin,
            })),
        );

        const CHUNK = 5000;
        for (let i = 0; i < items.length; i += CHUNK) {
          await tx.salaryCompositionItem.createMany({ data: items.slice(i, i + CHUNK) });
        }

        // ─── Nazorat: yozilgan jami = hisoblangan jami ───
        const [rowSum, savedItems] = await Promise.all([
          tx.salaryCompositionRow.aggregate({
            where: { batchId: batch.id },
            _sum: { totalTiyin: true },
            _count: { _all: true },
          }),
          tx.salaryCompositionItem.count({ where: { batchId: batch.id } }),
        ]);

        if (
          (rowSum._sum.totalTiyin ?? 0n) !== grandTotal ||
          rowSum._count._all !== prepared.length ||
          savedItems !== itemCount
        ) {
          throw new BadRequestException('Saqlashda nazorat summasi mos kelmadi — hech narsa saqlanmadi');
        }

        return { batchId: batch.id, replacedPrevious: previousIds.length > 0 };
      },
      { timeout: 60_000, maxWait: 10_000 },
    );

    return {
      success: true,
      batchId: result.batchId,
      saved: prepared.length,
      replacedPrevious: result.replacedPrevious,
      totalTiyin: grandTotal.toString(),
      payrollTiyin: check.payrollTiyin,
      status,
      mismatchCount: check.mismatchCount,
    };
  }

  async cancel(batchId: string): Promise<{ success: true; period: string }> {
    const batch = await this.prisma.salaryCompositionBatch.findUnique({ where: { id: batchId } });
    if (!batch) throw new NotFoundException('Yuklash topilmadi');
    if (batch.cancelledAt) throw new BadRequestException('Bu yuklash allaqachon bekor qilingan');

    await this.prisma.$transaction([
      this.prisma.salaryCompositionItem.deleteMany({ where: { batchId } }),
      this.prisma.salaryCompositionRow.deleteMany({ where: { batchId } }),
      this.prisma.salaryCompositionBatch.update({
        where: { id: batchId },
        data: { cancelledAt: new Date() },
      }),
    ]);

    return { success: true, period: batch.period };
  }

  }

  /** Oy holati — ish haqi sahifasidagi belgi uchun (jonli hisoblanadi) */
  async check(period: string): Promise<CheckSummary> {
    this.assertPeriod(period);

    const [batch, payroll] = await Promise.all([this.activeBatch(period), this.payroll(period)]);
    if (!batch) return this.buildCheck(period, null, payroll);

    const rows = await this.prisma.salaryCompositionRow.findMany({
      where: { batchId: batch.id },
      select: { employeePinfl: true, fullName: true, totalTiyin: true },
    });

    const summary = await this.buildCheck(
      period,
      rows.map((r) => ({ pinfl: r.employeePinfl, fullName: r.fullName ?? '—', totalTiyin: r.totalTiyin })),
      payroll,
    );

    await this.syncBatchStatus(batch.id, batch.status, batch.mismatchCount, batch.payrollTiyin, summary);
    return summary;
  }

  /** Soliq hisoboti keyin o'zgargan bo'lsa — saqlangan holatni yangilaymiz */
  private async syncBatchStatus(
    batchId: string,
    status: SalaryCompositionStatus,
    mismatchCount: number,
    payrollTiyin: bigint,
    summary: CheckSummary,
  ): Promise<void> {
    const nextStatus: SalaryCompositionStatus =
      summary.state === 'MATCHED' ? 'MATCHED' : summary.state === 'MISMATCH' ? 'MISMATCH' : 'NO_PAYROLL';
    const nextPayroll = BigInt(summary.payrollTiyin);

    if (nextStatus !== status || summary.mismatchCount !== mismatchCount || nextPayroll !== payrollTiyin) {
      await this.prisma.salaryCompositionBatch.update({
        where: { id: batchId },
        data: { status: nextStatus, mismatchCount: summary.mismatchCount, payrollTiyin: nextPayroll },
      });
    }
  }

  /** Oy tarkibi: xodimlar, komponentlar, jamilar va tekshiruv */
  async view(period: string): Promise<CompositionView> {
    this.assertPeriod(period);

    const [batch, payroll, components, usage] = await Promise.all([
      this.activeBatch(period),
      this.payroll(period),
      this.allComponents(),
      this.usageCounts(),
    ]);

    if (!batch) {
      return {
        period,
        batch: null,
        components: components
          .filter((c) => c.isActive)
          .map((c) => this.toComponentDto(c, usage.get(c.code) ?? 0)),
        byComponent: [],
        byCategory: [],
        check: await this.buildCheck(period, null, payroll),
        rows: [],
      };
    }

    const dbRows = await this.prisma.salaryCompositionRow.findMany({
      where: { batchId: batch.id },
      include: { items: { select: { componentCode: true, amountTiyin: true } } },
    });

    const registry = await this.registry(dbRows.map((r) => r.employeePinfl));

    const withItems = dbRows.map((r) => ({
      row: r,
      items: new Map(r.items.map((i) => [i.componentCode, i.amountTiyin])),
    }));

    const usedCodes = new Set(withItems.flatMap((r) => [...r.items.keys()]));
    const visible = components.filter((c) => c.isActive || usedCodes.has(c.code));

    const rows: CompositionRowDto[] = withItems
      .map(({ row, items }) => {
        const reg = registry.get(row.employeePinfl);
        const state = this.rowState(row.employeePinfl, row.totalTiyin, payroll);
        return {
          rowIndex: null,
          pinfl: row.employeePinfl,
          fullName: reg?.fullName ?? row.fullName ?? '—',
          inRegistry: !!reg,
          unitName: reg?.unitName ?? null,
          workedDays: row.workedDays,
          rate: row.rate,
          items: this.mapToRecord(items),
          totalTiyin: row.totalTiyin.toString(),
          payrollTiyin: state.payrollTiyin === null ? null : state.payrollTiyin.toString(),
          diffTiyin: state.diffTiyin === null ? null : state.diffTiyin.toString(),
          state: state.state,
        };
      })
      .sort((a, b) => a.fullName.localeCompare(b.fullName));

    const check = await this.buildCheck(
      period,
      rows.map((r) => ({ pinfl: r.pinfl, fullName: r.fullName, totalTiyin: BigInt(r.totalTiyin) })),
      payroll,
    );
    await this.syncBatchStatus(batch.id, batch.status, batch.mismatchCount, batch.payrollTiyin, check);

    const { byComponent, byCategory } = this.totals(withItems, components);
    const [batchDto] = await this.toBatchDtos([
      { ...batch, status: check.state === 'MATCHED' ? 'MATCHED' : check.state === 'MISMATCH' ? 'MISMATCH' : 'NO_PAYROLL', mismatchCount: check.mismatchCount },
    ]);

    return {
      period,
      batch: batchDto ?? null,
      components: visible.map((c) => this.toComponentDto(c, usage.get(c.code) ?? 0)),
      byComponent,
      byCategory,
      check,
      rows,
    };
  }

  /** Bitta xodimning oylar bo'yicha tarkibi */
  async employeeHistory(pinfl: string, from?: string, to?: string): Promise<EmployeeHistory> {
    if (!/^\d{14}$/.test(pinfl ?? '')) throw new BadRequestException('PINFL noto\'g\'ri');
    if (from) this.assertPeriod(from);
    if (to) this.assertPeriod(to);

    const periodFilter: { gte?: string; lte?: string } = {};
    if (from) periodFilter.gte = from;
    if (to) periodFilter.lte = to;

    const [dbRows, components, registry] = await Promise.all([
      this.prisma.salaryCompositionRow.findMany({
        where: {
          employeePinfl: pinfl,
          batch: { cancelledAt: null },
          ...(from || to ? { period: periodFilter } : {}),
        },
        include: { items: { select: { componentCode: true, amountTiyin: true } } },
        orderBy: { period: 'asc' },
      }),
      this.allComponents(),
      this.registry([pinfl]),
    ]);

    const reg = registry.get(pinfl);
    if (!reg && dbRows.length === 0) throw new NotFoundException('Xodim topilmadi');

    const periods = [...new Set(dbRows.map((r) => r.period))];
    const payrolls = new Map<string, PayrollInfo>();
    for (const period of periods) {
      payrolls.set(period, await this.payroll(period));
    }

    const withItems = dbRows.map((r) => ({
      row: r,
      items: new Map(r.items.map((i) => [i.componentCode, i.amountTiyin])),
    }));
    const usedCodes = new Set(withItems.flatMap((r) => [...r.items.keys()]));

    const { byCategory } = this.totals(withItems, components);

    return {
      employee: {
        pinfl,
        fullName: reg?.fullName ?? dbRows[0]?.fullName ?? '—',
        unitName: reg?.unitName ?? null,
        inRegistry: !!reg,
      },
      components: components.filter((c) => usedCodes.has(c.code)).map((c) => this.toComponentDto(c)),
      periods: withItems.map(({ row, items }) => {
        const payroll = payrolls.get(row.period) ?? { uploaded: false, map: new Map<string, bigint>() };
        const state = this.rowState(pinfl, row.totalTiyin, payroll);
        return {
          period: row.period,
          workedDays: row.workedDays,
          rate: row.rate,
          items: this.mapToRecord(items),
          totalTiyin: row.totalTiyin.toString(),
          payrollTiyin: state.payrollTiyin === null ? null : state.payrollTiyin.toString(),
          diffTiyin: state.diffTiyin === null ? null : state.diffTiyin.toString(),
          state: state.state,
        };
      }),
      byCategory,
    };
  }

  }
